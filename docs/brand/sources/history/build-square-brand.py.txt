from pathlib import Path
import json, xml.etree.ElementTree as ET

WORK=Path(__file__).resolve().parent
OUT=WORK.parent/'fatfish-notes-logo'
outline=json.loads((WORK/'brand-outlines.json').read_text(encoding='utf-8-sig'))
ns={'s':'http://www.w3.org/2000/svg'}
INK='#26362F'; PAPER='#F7F4EB'; SAGE='#B7C9B3'
layout={}

def wordmark(key,width,top,fill):
    item=outline[key]; b=item['bounds']; scale=width/b['width']
    x=(1024-width)/2
    layout[key]={'text':item['text'],'font_source':item['family'],'bounds':[x,top,x+width,top+b['height']*scale],'tracking_em':item['tracking']/item['em'],'path_points':item['points']}
    return f'<g id="wordmark-{key}" fill="{fill}" transform="translate({x:.6f} {top:.6f}) scale({scale:.9f}) translate({-b["x"]:.6f} {-b["y"]:.6f})"><path fill-rule="{item.get("fill_rule","nonzero")}" d="{item["d"]}"/></g>'

for dark in [False,True]:
    theme='dark' if dark else 'light'; fill=PAPER if dark else INK; bg=INK if dark else PAPER
    master=ET.parse(OUT/'svg'/('fatfish-mark-dark.svg' if dark else 'fatfish-mark.svg')).getroot()
    srcpaths=master.findall('.//s:path',ns)
    assert len(srcpaths)==2
    mark='\n'.join('<path '+ ' '.join(f'{k}="{v}"' for k,v in p.attrib.items())+'/>' for p in srcpaths)
    cn=wordmark('chinese',760,638,fill)
    en=wordmark('english',560,838,fill)
    title=f'肥鱼笔记 / Fatfish Notes · {"深底" if dark else "浅底"}正方形品牌组合版'
    svg=f'''<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024" viewBox="0 0 1024 1024" role="img" aria-label="{title}">
<title>{title}</title>
<desc>叠页鱼形标志、肥鱼笔记与 Fatfish Notes。所有可见文字均已转为矢量路径。</desc>
<rect width="1024" height="1024" fill="{bg}"/>
<g id="fish-symbol" transform="translate(80 34) scale(1.08)">
  <g transform="translate(-219 -54)">
    {mark}
  </g>
</g>
{cn}
{en}
</svg>
'''
    (OUT/'svg'/f'fatfish-brand-square-{theme}.svg').write_text(svg,encoding='utf-8')

layout['symbol']={'source':'fatfish-mark.svg / fatfish-mark-dark.svg','unchanged_paths':True,'transform':'translate(80 34) scale(1.08)'}
layout['canvas']=[1024,1024]
layout['png_sizes']=[256,512,1024,2048]
(WORK/'brand-square-layout.json').write_text(json.dumps(layout,ensure_ascii=False,indent=2),encoding='utf-8')
