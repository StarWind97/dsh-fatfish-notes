from pathlib import Path
import json, html

WORK=Path(__file__).resolve().parent

g=json.loads((WORK/'geometry.json').read_text(encoding='utf-8'))
INK='#26362F'; SAGE='#B7C9B3'; PAPER='#F7F4EB'; SMALL_SAGE='#809D7B'

def paths(small=False,dark=False,mono=False):
    front=PAPER if dark else INK
    rear=front if mono else (SAGE if dark or not small else SMALL_SAGE)
    cx,cy=g['eye']['cx'],g['eye']['cy']; r=28 if small else g['eye']['r']
    hole=f'M {cx+r} {cy} A {r} {r} 0 1 0 {cx-r} {cy} A {r} {r} 0 1 0 {cx+r} {cy} Z'
    shift=' transform="translate(-6 -16)"' if small else ''
    return f'<path fill="{rear}"{shift} d="{g["rear"]["d"]}"/>\n  <path fill="{front}" fill-rule="evenodd" d="{g["front"]["d"]} {hole}"/>'

variants={}
for small in [False,True]:
    for kind in ['','-dark','-mono']:
        name='fatfish-mark'+('-small' if small else '')+kind
        p=paths(small,kind=='-dark',kind=='-mono')
        title='肥鱼笔记 · 叠页标志'+(' · 小尺寸' if small else '')
        svg=f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 800 560" width="800" height="560" role="img" aria-label="{title}">\n<title>{title}</title>\n<g transform="translate(-219 -54)">\n  {p}\n</g>\n</svg>\n'
        (OUT/'svg'/(name+'.svg')).write_text(svg,encoding='utf-8')
        variants[name]={'viewBox':[800,560],'small':small,'kind':kind or 'color','type':'mark'}
    for dark in [False,True]:
        name='fatfish-icon'+('-small' if small else '')+('-dark' if dark else '-light')
        p=paths(small,dark)
        bg=INK if dark else PAPER
        title='肥鱼笔记 · '+('深色' if dark else '浅色')+'应用图标'+(' · 小尺寸' if small else '')
        svg=f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 880 880" width="880" height="880" role="img" aria-label="{title}">\n<title>{title}</title>\n<rect width="880" height="880" rx="192" fill="{bg}"/>\n<g transform="translate(-179 106)">\n  {p}\n</g>\n</svg>\n'
        (OUT/'svg'/(name+'.svg')).write_text(svg,encoding='utf-8')
        variants[name]={'viewBox':[880,880],'small':small,'kind':'dark' if dark else 'light','type':'icon'}

# Source-space proof masks make contour fidelity measurable independently of the presentation layout.
for part in ['front','rear']:
    d=g[part]['d']
    if part=='front':
        cx,cy,r=g['eye']['cx'],g['eye']['cy'],g['eye']['r']
        d+=f' M {cx+r} {cy} A {r} {r} 0 1 0 {cx-r} {cy} A {r} {r} 0 1 0 {cx+r} {cy} Z'
    (WORK/(part+'-vector.svg')).write_text(f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 1254 1254" width="1254" height="1254"><path fill="black" fill-rule="evenodd" d="{d}"/></svg>',encoding='utf-8')

(WORK/'variants.json').write_text(json.dumps(variants,indent=2),encoding='utf-8')

sizes=[16,20,24,32,48,64]
def size_row(kind,dark=False):
    cells=[]
    for size in sizes:
        suffix='-small' if size<=24 else ''
        name=('fatfish-mark'+suffix+('-dark' if dark else '')) if kind=='mark' else ('fatfish-icon'+suffix+('-dark' if dark else '-light'))
        cells.append(f'<div class="size"><div class="slot"><img src="svg/{name}.svg" width="{size}" height="{size}" alt="{size}px 标志"></div><span>{size} px</span></div>')
    return '<div class="sizes">'+''.join(cells)+'</div>'
