const fs = require('node:fs');
const path = require('node:path');
const {pathToFileURL} = require('node:url');
const {chromium} = require('D:/ProgramFiles/nodejs/node_cache/_npx/31e32ef8478fbf80/node_modules/playwright');
const work=__dirname;
const out=path.join(work,'..','fatfish-notes-logo');
const sizes=[16,20,24,32,48,64,128,180,256,512];

(async()=>{
 const browser=await chromium.launch({executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe',headless:true});
 try{
  const page=await browser.newPage({viewport:{width:1120,height:900},deviceScaleFactor:1});
  async function raster(file,w,h,destination){
   const svg=fs.readFileSync(file,'utf8');
   const data=await page.evaluate(async({svg,w,h})=>{
    const image=new Image(); image.src='data:image/svg+xml;base64,'+btoa(unescape(encodeURIComponent(svg)));
    await image.decode();
    const canvas=document.createElement('canvas'); canvas.width=w; canvas.height=h;
    const ctx=canvas.getContext('2d');
    const scale=Math.min(w/image.naturalWidth,h/image.naturalHeight);
    const dw=image.naturalWidth*scale,dh=image.naturalHeight*scale;
    ctx.drawImage(image,(w-dw)/2,(h-dh)/2,dw,dh);
    return canvas.toDataURL('image/png').split(',')[1];
   },{svg,w,h});
   fs.writeFileSync(destination,Buffer.from(data,'base64'));
  }
  for(const size of sizes){
   for(const [type,color] of [['mark',''],['mark','-dark'],['mark','-mono'],['icon','-light'],['icon','-dark']]){
    const small=size<=24?'-small':'';
    const name=`fatfish-${type}${small}${color}`;
    const png=`fatfish-${type}${color}-${size}.png`;
    const height=type==='mark'?Math.round(size*560/800):size;
    await raster(path.join(out,'svg',name+'.svg'),size,height,path.join(out,'png',png));
   }
  }
  for(const [name,dest] of [['fatfish-icon-light','standard-16'],['fatfish-icon-small-light','optical-16']])
   await raster(path.join(out,'svg',name+'.svg'),16,16,path.join(out,'proof',dest+'.png'));
  for(const part of ['front','rear'])
   await raster(path.join(work,part+'-vector.svg'),1254,1254,path.join(work,part+'-vector.png'));
  const all=JSON.parse(fs.readFileSync(path.join(work,'variants.json'),'utf8'));
  for(const [name,v] of Object.entries(all)){
   await raster(path.join(out,'svg',name+'.svg'),v.viewBox[0],v.viewBox[1],path.join(work,name+'-native.png'));
  }
  await page.goto(pathToFileURL(path.join(out,'preview.html')).href);
  await page.waitForFunction(()=>[...document.images].every(x=>x.complete&&x.naturalWidth>0));
  await page.screenshot({path:path.join(out,'proof','preview-desktop.png'),fullPage:true});
  const findings=await page.evaluate(()=>({images:[...document.images].length,broken:[...document.images].filter(x=>!x.complete||!x.naturalWidth).length,overflow:document.documentElement.scrollWidth>innerWidth,dpr:devicePixelRatio}));
  console.log(JSON.stringify({browser:browser.version(),pngs:fs.readdirSync(path.join(out,'png')).length,preview:findings}));
 }finally{await browser.close();}
})().catch(e=>{console.error(e);process.exitCode=1;});
