def subpixel_loops(crop):
    gray=crop.convert('L').filter(ImageFilter.GaussianBlur(.35))
    px=gray.load(); w,h=gray.size; threshold=147.5
    positions={}; graph={}
    def link(a,b):
        graph.setdefault(a,[]).append(b);graph.setdefault(b,[]).append(a)
    for y in range(h-1):
        for x in range(w-1):
            vals=[px[x,y],px[x+1,y],px[x+1,y+1],px[x,y+1]]
            corners=[(x+.5,y+.5),(x+1.5,y+.5),(x+1.5,y+1.5),(x+.5,y+1.5)]
            keys=[('h',x,y),('v',x+1,y),('h',x,y+1),('v',x,y)]
            crossings=[]
            for e in range(4):
                j=(e+1)%4
                if (vals[e]<threshold)==(vals[j]<threshold):continue
                key=keys[e]; t=(threshold-vals[e])/(vals[j]-vals[e])
                positions[key]=add(corners[e],mul(sub(corners[j],corners[e]),t))
                crossings.append(e)
            if len(crossings)==2:link(keys[crossings[0]],keys[crossings[1]])
            elif len(crossings)==4:
                pairs=[(0,1),(2,3)] if (vals[0]<threshold)==(sum(vals)/4<threshold) else [(0,3),(1,2)]
                for a,b in pairs:link(keys[a],keys[b])
    unseen=set(graph); result=[]
    while unseen:
        start=min(unseen); prev=None; cur=start; loop=[]
        while cur in unseen:
            unseen.remove(cur);loop.append(positions[cur])
            adj=graph[cur]
            assert len(adj)==2, (cur,adj)
            nxt=adj[0] if adj[0]!=prev else adj[1]
            prev,cur=cur,nxt
        assert cur==start
        area=abs(sum(loop[i][0]*loop[(i+1)%len(loop)][1]-loop[(i+1)%len(loop)][0]*loop[i][1] for i in range(len(loop)))/2)
        if area>1.5:result.append(loop)
    return sorted(result,key=lambda ps:min(p[0] for p in ps))

def outline_loop(loop,tolerance):
    # Start on a straight run and keep the start tangent continuous across closure.
    n=len(loop)
    def curvature(i):
        v1=unit(sub(loop[i],loop[(i-4)%n]));v2=unit(sub(loop[(i+4)%n],loop[i]))
        return 1-dot(v1,v2)
    start=min(range(n),key=curvature)
    loop=loop[start:]+loop[:start]
    tangent=unit(sub(loop[3],loop[-3]))
    # Retain dense boundary samples so a curve cannot bulge between sparse
    # endpoints on the straight inner strokes of the Chinese characters.
    pts=loop+[loop[0]]
    curves=fit(pts,tangent,mul(tangent,-1),tolerance)
    def fmt(p):return f'{p[0]:.3f} {p[1]:.3f}'
    d='M '+fmt(curves[0][0])
    for curve in curves:d+=' C '+' '.join(fmt(p) for p in curve[1:])
    return d+' Z',len(curves)
