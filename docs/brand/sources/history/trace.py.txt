def add(a,b): return (a[0]+b[0], a[1]+b[1])
def sub(a,b): return (a[0]-b[0], a[1]-b[1])
def mul(a,s): return (a[0]*s, a[1]*s)
def dot(a,b): return a[0]*b[0]+a[1]*b[1]
def length(a): return math.hypot(*a)
def unit(a):
    d=length(a)
    return mul(a,1/d) if d else (0.,0.)

def contours(mask):
    edges={}
    for x,y in mask:
        if (x,y-1) not in mask: edges[(x,y)]=(x+1,y)
        if (x+1,y) not in mask: edges[(x+1,y)]=(x+1,y+1)
        if (x,y+1) not in mask: edges[(x+1,y+1)]=(x,y+1)
        if (x-1,y) not in mask: edges[(x,y+1)]=(x,y)
    result=[]
    while edges:
        start=next(iter(edges)); current=start; chain=[]
        while current in edges:
            chain.append(current); current=edges.pop(current)
            if current==start: break
        if len(chain)>20: result.append(chain)
    return sorted(result,key=len,reverse=True)

def largest_component(mask):
    unseen=set(mask); largest=set()
    while unseen:
        start=unseen.pop(); group={start}; stack=[start]
        while stack:
            x,y=stack.pop()
            for p in [(x-1,y),(x+1,y),(x,y-1),(x,y+1)]:
                if p in unseen: unseen.remove(p); group.add(p); stack.append(p)
        if len(group)>len(largest): largest=group
    return largest

def dist_line(p,a,b):
    ab=sub(b,a); d=dot(ab,ab)
    if not d: return length(sub(p,a))
    t=max(0,min(1,dot(sub(p,a),ab)/d))
    return length(sub(p,add(a,mul(ab,t))))

def rdp(points,epsilon):
    if len(points)<3: return points
    maximum,index=max((dist_line(p,points[0],points[-1]),i) for i,p in enumerate(points[1:-1],1))
    if maximum<=epsilon: return [points[0],points[-1]]
    return rdp(points[:index+1],epsilon)[:-1]+rdp(points[index:],epsilon)

def bezier(q,t):
    s=1-t
    return add(add(mul(q[0],s*s*s),mul(q[1],3*t*s*s)),add(mul(q[2],3*t*t*s),mul(q[3],t*t*t)))

def generate(points,u,t1,t2):
    p0,p3=points[0],points[-1]
    c00=c01=c11=x0=x1=0.
    for p,t in zip(points,u):
        s=1-t; b0=s*s*s; b1=3*t*s*s; b2=3*t*t*s; b3=t*t*t
        a0=mul(t1,b1); a1=mul(t2,b2)
        tmp=sub(p,add(mul(p0,b0+b1),mul(p3,b2+b3)))
        c00+=dot(a0,a0); c01+=dot(a0,a1); c11+=dot(a1,a1)
        x0+=dot(a0,tmp); x1+=dot(a1,tmp)
    determinant=c00*c11-c01*c01
    a=(x0*c11-x1*c01)/determinant if abs(determinant)>1e-9 else -1
    b=(c00*x1-c01*x0)/determinant if abs(determinant)>1e-9 else -1
    d=length(sub(p3,p0))
    if a<1e-5*d or b<1e-5*d or a>2*d or b>2*d: a=b=d/3
    return [p0,add(p0,mul(t1,a)),add(p3,mul(t2,b)),p3]

def fit(points,t1,t2,error):
    if len(points)==2:
        d=length(sub(points[1],points[0]))/3
        return [[points[0],add(points[0],mul(t1,d)),add(points[1],mul(t2,d)),points[1]]]
    u=[0.]
    for i in range(1,len(points)): u.append(u[-1]+length(sub(points[i],points[i-1])))
    u=[x/u[-1] for x in u]
    curve=generate(points,u,t1,t2)
    maximum,index=max((length(sub(bezier(curve,u[i]),points[i])),i) for i in range(1,len(points)-1))
    if maximum<=error: return [curve]
    center=unit(sub(points[index-1],points[index+1]))
    return fit(points[:index+1],t1,center,error)+fit(points[index:],mul(center,-1),t2,error)

def smooth_path(contour,landmarks):
    # Trace the chosen raster at subpixel tolerance, then fit editable cubic curves.
    indices=sorted({min(range(len(contour)),key=lambda i:length(sub(contour[i],p))) for p in landmarks})
    curves=[]
    for j,start in enumerate(indices):
        end=indices[(j+1)%len(indices)]
        section=contour[start:end+1] if end>start else contour[start:]+contour[:end+1]
        pts=rdp(section,0.7)
        t1=unit(sub(section[min(10,len(section)-1)],section[0]))
        t2=unit(sub(section[max(0,len(section)-11)],section[-1]))
        curves.extend(fit(pts,t1,t2,2.0))
    def f(p): return f'{p[0]:.2f} {p[1]:.2f}'
    path='M '+f(curves[0][0])
    for c in curves: path+=' C '+f(c[1])+' '+f(c[2])+' '+f(c[3])
    return path+' Z',curves
